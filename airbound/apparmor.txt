#include <tunables/global>

# What each program inside AirBound may touch. Without this file the add-on runs under
# Docker's general profile, where every program in the container can read every file in it.
#
# The programs that face the internet (Caddy, the relay client frpc and CrowdSec) each get
# a profile of their own with only their own files. So a hole in one of them does not give
# away AirBound's settings, the keys of the others, the copy of Home Assistant's
# configuration folder, or AirBound's key to Home Assistant, which sits in the memory and
# environment of the main program.
#
# tests/unit/test_apparmor.py checks this file against the code, and the test round checks
# on a real Home Assistant that it is in force and that nothing was refused.

profile airbound flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # ---- The add-on itself: Docker's init and Python ----

  /sbin/docker-init rix,
  /usr/local/bin/python3* rix,
  /usr/local/lib/** mr,
  /usr/lib/** mr,
  /lib/** mr,

  # The code is read, never changed. (Python would like to save its compiled form; it does
  # without, as quietly as "deny" makes it.)
  /app/ r,
  /app/** r,
  deny /app/** w,
  deny /usr/local/lib/** w,

  # AirBound's own folder: settings, keys, logs and what the helper programs keep.
  /data/ r,
  /data/** rwk,

  # From Home Assistant's configuration folder: who has two-step login, and Home Assistant's own bans.
  /homeassistant/.storage/auth_module.totp r,
  /homeassistant/.storage/auth_module.notify r,
  /homeassistant/ip_bans.yaml r,

  # CrowdSec's starting files in the image, copied to /data the first time.
  /var/lib/crowdsec-default/ r,
  /var/lib/crowdsec-default/** r,

  /etc/** r,
  /usr/share/zoneinfo/** r,
  /usr/share/ca-certificates/** r,
  /tmp/ r,
  owner /tmp/** rwk,
  /dev/null rw,
  /dev/urandom r,
  @{PROC}/@{pid}/fd/ r,
  @{PROC}/@{pid}/attr/current r,
  @{PROC}/@{pid}/{cgroup,mounts,mountinfo,stat,status} r,
  @{PROC}/sys/kernel/** r,
  @{PROC}/sys/net/core/somaxconn r,
  /sys/fs/cgroup/** r,
  /sys/devices/system/cpu/** r,

  # Ordinary network connections only: no raw packets.
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  network netlink raw,

  # Stopping and reloading the helper programs, and being stopped by Docker.
  signal,

  # The helper programs, each under its own profile below.
  /usr/local/bin/caddy cx -> caddy,
  /usr/local/bin/frpc cx -> frpc,
  /usr/local/bin/crowdsec cx -> crowdsec,
  /usr/local/bin/cscli cx -> crowdsec,

  # ---- Caddy: answers on port 443, holds the certificates ----

  profile caddy flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    /usr/local/bin/caddy mr,
    /data/Caddyfile r,
    /data/caddy/ rw,
    /data/caddy/** rwk,
    /data/logs/ r,
    /data/logs/access* rw,

    /etc/** r,
    /usr/share/zoneinfo/** r,
    /usr/share/ca-certificates/** r,
    /dev/null rw,
    /dev/urandom r,
    @{PROC}/ r,
    @{PROC}/@{pid}/ r,
    @{PROC}/@{pid}/fd/ r,
    @{PROC}/@{pid}/{cgroup,mountinfo,maps,stat,status,limits,net/netstat} r,
    @{PROC}/{cpuinfo,meminfo,stat,loadavg,uptime} r,
    @{PROC}/sys/kernel/** r,
    @{PROC}/sys/net/core/somaxconn r,
    /sys/kernel/mm/transparent_hugepage/hpage_pmd_size r,
    /sys/kernel/mm/hugepages/ r,
    /sys/fs/cgroup/** r,
    /sys/devices/system/cpu/** r,
    /usr/lib/os-release r,
    /.dockerenv r,
    # These programs look up what kind of machine they run on, and for that try to read the
    # environment of the first program in the container: the one place AirBound's key to
    # Home Assistant can be read. They work without, so it is refused, quietly.
    deny @{PROC}/@{pid}/environ r,
    deny @{PROC}/@{pid}/mem rw,
    deny ptrace,
    deny @{PROC}/modules r,
    deny @{PROC}/bus/** r,
    deny /sys/devices/virtual/dmi/** r,

    network inet stream,
    network inet dgram,
    network inet6 stream,
    network inet6 dgram,
    network netlink raw,
    capability net_bind_service,

    signal (receive),
  }

  # ---- frpc: the connection out to the relay ----

  profile frpc flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    /usr/local/bin/frpc mr,
    /data/frpc.toml r,
    /data/relay/ r,
    /data/relay/** r,
    /app/airbound/hosted_relay_ca.pem r,

    /etc/** r,
    /usr/share/zoneinfo/** r,
    /usr/share/ca-certificates/** r,
    /dev/null rw,
    /dev/urandom r,
    @{PROC}/ r,
    @{PROC}/@{pid}/ r,
    @{PROC}/@{pid}/fd/ r,
    @{PROC}/@{pid}/{cgroup,mountinfo,maps,stat,status,limits,net/netstat} r,
    @{PROC}/{cpuinfo,meminfo,stat,loadavg,uptime} r,
    @{PROC}/sys/kernel/** r,
    @{PROC}/sys/net/core/somaxconn r,
    /sys/kernel/mm/transparent_hugepage/hpage_pmd_size r,
    /sys/kernel/mm/hugepages/ r,
    /sys/fs/cgroup/** r,
    /sys/devices/system/cpu/** r,
    /usr/lib/os-release r,
    /.dockerenv r,
    # These programs look up what kind of machine they run on, and for that try to read the
    # environment of the first program in the container: the one place AirBound's key to
    # Home Assistant can be read. They work without, so it is refused, quietly.
    deny @{PROC}/@{pid}/environ r,
    deny @{PROC}/@{pid}/mem rw,
    deny ptrace,
    deny @{PROC}/modules r,
    deny @{PROC}/bus/** r,
    deny /sys/devices/virtual/dmi/** r,

    network inet stream,
    network inet dgram,
    network inet6 stream,
    network inet6 dgram,
    network netlink raw,

    signal (receive),
  }

  # ---- CrowdSec and its command line tool: read the logs, keep the ban list ----

  profile crowdsec flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # cscli starts crowdsec itself for some of its commands; both stay in this profile.
    /usr/local/bin/crowdsec mrix,
    /usr/local/bin/cscli mrix,
    /data/crowdsec/ rw,
    /data/crowdsec/** rwlk,
    /data/logs/ r,
    /data/logs/** r,
    /usr/local/lib/crowdsec/plugins/ r,

    /etc/** r,
    /usr/share/zoneinfo/** r,
    /usr/share/ca-certificates/** r,
    /tmp/ r,
    owner /tmp/** rwk,
    /dev/null rw,
    /dev/urandom r,
    @{PROC}/ r,
    @{PROC}/@{pid}/ r,
    @{PROC}/@{pid}/fd/ r,
    @{PROC}/@{pid}/{cgroup,mountinfo,maps,stat,status,limits,net/netstat} r,
    @{PROC}/{cpuinfo,meminfo,stat,loadavg,uptime} r,
    @{PROC}/sys/kernel/** r,
    @{PROC}/sys/net/core/somaxconn r,
    /sys/kernel/mm/transparent_hugepage/hpage_pmd_size r,
    /sys/kernel/mm/hugepages/ r,
    /sys/fs/cgroup/** r,
    /sys/devices/system/cpu/** r,
    /usr/lib/os-release r,
    /.dockerenv r,
    # These programs look up what kind of machine they run on, and for that try to read the
    # environment of the first program in the container: the one place AirBound's key to
    # Home Assistant can be read. They work without, so it is refused, quietly.
    deny @{PROC}/@{pid}/environ r,
    deny @{PROC}/@{pid}/mem rw,
    deny ptrace,
    deny @{PROC}/modules r,
    deny @{PROC}/bus/** r,
    deny /sys/devices/virtual/dmi/** r,

    network inet stream,
    network inet dgram,
    network inet6 stream,
    network inet6 dgram,
    network netlink raw,

    signal (receive),
  }
}
